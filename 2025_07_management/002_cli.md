## 📁 Cấu trúc CLI (file `main.py`)

```python
# main.py
import typer
from pathlib import Path
import json

app = typer.Typer()

# Load danh sách chức năng từ modules/
MODULE_DIR = Path("modules")
PROJECT_DIR = Path("projects")

@app.command()
def list_projects():
    """Liệt kê các project hiện có"""
    projects = [p.name for p in PROJECT_DIR.iterdir() if p.is_dir()]
    for prj in projects:
        typer.echo(f"- {prj}")

@app.command()
def list_features():
    """Liệt kê các tính năng (modules) hiện có"""
    for mod in MODULE_DIR.iterdir():
        if mod.is_dir():
            typer.echo(f"- {mod.name}")

@app.command()
def run_feature(feature: str):
    """Chạy một chức năng cụ thể"""
    mod_path = MODULE_DIR / feature
    config_path = mod_path / "config.json"
    if not config_path.exists():
        typer.echo(f"[❌] Không tìm thấy config cho {feature}")
        raise typer.Exit()

    config = json.loads(config_path.read_text())
    project = config["project"]
    db_index = config["db_index"]

    typer.echo(f"[🚀] Đang chạy {feature}")
    typer.echo(f"→ Dữ liệu từ project: {project}")
    typer.echo(f"→ Sử dụng index: {db_index}")

    # Ví dụ: import và chạy logic
    try:
        logic_module = __import__(f"modules.{feature}.logic", fromlist=["run"])
        logic_module.run(config)
    except Exception as e:
        typer.echo(f"[❌] Lỗi khi chạy chức năng: {e}")
        raise typer.Exit()

if __name__ == "__main__":
    app()
```

## ✅ Cách sử dụng CLI:

```bash
# Hiển thị danh sách project
python main.py list-projects

# Hiển thị các chức năng khả dụng
python main.py list-features

# Chạy một chức năng cụ thể
python main.py run-feature feature1_chuc_nang_1
```

---

## ⏭️ Gợi ý mở rộng:

* Thêm option `--project`, `--version` để ghi đè config
* Thêm lệnh `init-feature` để tạo module mới kèm config tự động
* Hỗ trợ `chat-ai` để mapping từ câu lệnh sang enum rồi tự gọi `run-feature`
